Sure! Here's a warm-themed button component:
@Composable
fun WarmButton(
  text: String,
  onClick: () -> Unit
) {
  Button(
    onClick = onClick,
    colors = ButtonDefaults
      .buttonColors(
        backgroundColor = Color(0xFFF4845F)
      )
  ) {
    Text(text)
  }
}
